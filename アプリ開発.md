## 🧠 仕様まとめ

### 🎨 お絵描き画面（drawing_canvas.dart）
- 指で描ける
- 色変更できる（9色）
- SE（tap / clear）が鳴る
- 全部消すボタン
- ぷにぷにアニメ画面へ遷移するボタン
- 背景はパステルクリーム
- UI はパステルピンク中心

---

### 🎬 ぷにぷにアニメ画面（puni_animation.dart）
- 描いた絵を受け取る
- ScaleTransition で “ぷにぷに” アニメ
- 戻るボタン
- 保存ボタン（今はダミー）
- 背景はパステルクリーム

---

### 🔊 SE（sound_manager.dart）
- `tap.mp3` → 色変更時
- `clear.mp3` → 全消し時
- audioplayers を使用

---

### 🖍 線データ（stroke.dart）
- points（Offset の配列）
- color
- strokeWidth

### 4. lib/drawing_canvas.dart

```dart
import 'package:flutter/material.dart';
import 'stroke.dart';
import 'sound_manager.dart';
import 'puni_animation.dart';

class DrawingCanvasPage extends StatefulWidget {
  const DrawingCanvasPage({super.key});

  @override
  State<DrawingCanvasPage> createState() => _DrawingCanvasPageState();
}

class _DrawingCanvasPageState extends State<DrawingCanvasPage> {
  final List<DrawnLine> _lines = [];
  Color _currentColor = const Color(0xFFFFAACC);
  final double _strokeWidth = 8.0;

  void _onPanStart(DragStartDetails details) {
    final box = context.findRenderObject() as RenderBox;
    final point = box.globalToLocal(details.globalPosition);

    setState(() {
      _lines.add(
        DrawnLine(
          points: [point],
          color: _currentColor,
          strokeWidth: _strokeWidth,
        ),
      );
    });
  }

  void _onPanUpdate(DragUpdateDetails details) {
    final box = context.findRenderObject() as RenderBox;
    final point = box.globalToLocal(details.globalPosition);

    setState(() {
      _lines.last.points.add(point);
    });
  }

  void _clearCanvas() {
    setState(() {
      _lines.clear();
    });
    SoundManager.clear();
  }

  void _changeColor(Color color) {
    setState(() {
      _currentColor = color;
    });
    SoundManager.tap();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFFFF8E8),
      appBar: AppBar(
        backgroundColor: const Color(0xFFFFCCE5),
        elevation: 0,
        title: const Text('ぷにぷにおえかき'),
        actions: [
          IconButton(
            icon: const Icon(Icons.delete_forever),
            onPressed: _clearCanvas,
          ),
          IconButton(
            icon: const Icon(Icons.play_circle_fill),
            onPressed: () {
              Navigator.push(
                context,
                MaterialPageRoute(
                  builder: (_) => PuniPuniAnimationPage(lines: _lines),
                ),
              );
            },
          ),
        ],
      ),
      body: Column(
        children: [
          Expanded(
            child: GestureDetector(
              onPanStart: _onPanStart,
              onPanUpdate: _onPanUpdate,
              child: CustomPaint(
                painter: _OekakiPainter(lines: _lines),
                child: Container(),
              ),
            ),
          ),
          const SizedBox(height: 8),
          _buildColorBar(),
          const SizedBox(height: 16),
        ],
      ),
    );
  }

  Widget _buildColorBar() {
    final colors = [
      const Color(0xFFFFAACC),
      const Color(0xFFFFDDAA),
      const Color(0xFFFFFFAA),
      const Color(0xFFAAFFCC),
      const Color(0xFFAAEEFF),
      const Color(0xFFCCAAFF),
      const Color(0xFFCC9966),
      const Color(0xFFFFFFFF),
      const Color(0xFF000000),
    ];

    return SizedBox(
      height: 60,
      child: ListView.separated(
        scrollDirection: Axis.horizontal,
        padding: const EdgeInsets.symmetric(horizontal: 16),
        itemCount: colors.length,
        separatorBuilder: (_, __) => const SizedBox(width: 8),
        itemBuilder: (context, index) {
          final color = colors[index];
          final isSelected = color == _currentColor;

          return GestureDetector(
            onTap: () => _changeColor(color),
            child: AnimatedContainer(
              duration: const Duration(milliseconds: 120),
              width: isSelected ? 46 : 40,
              height: isSelected ? 46 : 40,
              decoration: BoxDecoration(
                color: color,
                shape: BoxShape.circle,
                boxShadow: isSelected
                    ? [
                        BoxShadow(
                          color: color.withOpacity(0.6),
                          blurRadius: 8,
                          spreadRadius: 2,
                        ),
                      ]
                    : [],
              ),
            ),
          );
        },
      ),
    );
  }
}

class _OekakiPainter extends CustomPainter {
  _OekakiPainter({required this.lines});
  final List<DrawnLine> lines;

  @override
  void paint(Canvas canvas, Size size) {
    for (final line in lines) {
      final paint = Paint()
        ..color = line.color
        ..strokeWidth = line.strokeWidth
        ..strokeCap = StrokeCap.round
        ..style = PaintingStyle.stroke;

      for (int i = 0; i < line.points.length - 1; i++) {
        canvas.drawLine(line.points[i], line.points[i + 1], paint);
      }
    }
  }

  @override
  bool shouldRepaint(covariant _OekakiPainter oldDelegate) => true;
}
```

### 5. lib/puni_animation.dart

```dart
import 'package:flutter/material.dart';
import 'stroke.dart';

class PuniPuniAnimationPage extends StatefulWidget {
  const PuniPuniAnimationPage({super.key, required this.lines});

  final List<DrawnLine> lines;

  @override
  State<PuniPuniAnimationPage> createState() => _PuniPuniAnimationPageState();
}

class _PuniPuniAnimationPageState extends State<PuniPuniAnimationPage>
    with SingleTickerProviderStateMixin {

  late AnimationController _controller;
  late Animation<double> _scale;

  @override
  void initState() {
    super.initState();

    _controller = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 2),
    )..repeat(reverse: true);

    _scale = Tween<double>(begin: 0.95, end: 1.05).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeInOut),
    );
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  void _saveVideo() {
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text("保存機能は後で実装するよ！")),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFFFF8E8),
      appBar: AppBar(
        backgroundColor: const Color(0xFFFFCCE5),
        elevation: 0,
        leading: IconButton(
          icon: const Icon(Icons.arrow_back),
          onPressed: () => Navigator.pop(context),
        ),
        actions: [
          IconButton(
            icon: const Icon(Icons.save_alt),
            onPressed: _saveVideo,
          ),
        ],
      ),
      body: Center(
        child: ScaleTransition(
          scale: _scale,
          child: CustomPaint(
            painter: _PuniPainter(lines: widget.lines),
            child: const SizedBox(width: 300, height: 300),
          ),
        ),
      ),
    );
  }
}

class _PuniPainter extends CustomPainter {
  _PuniPainter({required this.lines});
  final List<DrawnLine> lines;

  @override
  void paint(Canvas canvas, Size size) {
    for (final line in lines) {
      final paint = Paint()
        ..color = line.color
        ..strokeWidth = line.strokeWidth
        ..strokeCap = StrokeCap.round
        ..style = PaintingStyle.stroke;

      for (int i = 0; i < line.points.length - 1; i++) {
        canvas.drawLine(line.points[i], line.points[i + 1], paint);
      }
    }
  }

  @override
  bool shouldRepaint(covariant _PuniPainter oldDelegate) => true;
}
```
